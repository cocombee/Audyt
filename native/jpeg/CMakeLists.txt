cmake_minimum_required(VERSION 3.16)
project(audyt_jpeg C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(PNG_SUPPORTED OFF CACHE BOOL "" FORCE)
set(WITH_TURBOJPEG OFF CACHE BOOL "" FORCE)
add_subdirectory(mozjpeg mozjpeg)
add_library(audyt_jpeg SHARED audyt_jpeg.c)
target_include_directories(audyt_jpeg PRIVATE mozjpeg ${CMAKE_CURRENT_BINARY_DIR}/mozjpeg)
target_link_libraries(audyt_jpeg PRIVATE jpeg-static)
set_target_properties(audyt_jpeg PROPERTIES C_VISIBILITY_PRESET hidden)
